账号数据库更新

November 2, 2026

前言#

    此次更新为账号数据库的更新。数据包括角色名、等级、体力、经验、金币等

数据库格式#

    这些储存在MySQL中。挂载到 MySQL 容器内部的/docker-entrypoint-initdb.d启用docker的时候自动建立。

-- 玩家角色表：基础信息，uid 关联 account.id
CREATE TABLE IF NOT EXISTS player (
    uid        BIGINT UNSIGNED PRIMARY KEY COMMENT '玩家ID = account.id',
    name       VARCHAR(64) NOT NULL DEFAULT '' COMMENT '角色名',
    level      INT         NOT NULL DEFAULT 1 COMMENT '等级',
    stamina    INT         NOT NULL DEFAULT 120 COMMENT '体力',
    exp        INT         NOT NULL DEFAULT 0 COMMENT '经验（战斗结算累积）',
    gold       INT         NOT NULL DEFAULT 0 COMMENT '金币（战斗结算累积）',
    created_at TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (uid) REFERENCES account (id)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

-- 战斗记录表：每场结算落一条，uid 关联 player.uid
CREATE TABLE IF NOT EXISTS battle_record (
    id           BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY COMMENT '记录ID',
    uid          BIGINT UNSIGNED NOT NULL COMMENT '玩家ID',
    result       INT         NOT NULL COMMENT '0 胜 / 1 败',
    score        INT         NOT NULL DEFAULT 0 COMMENT '结算分数',
    duration_sec INT         NOT NULL DEFAULT 0 COMMENT '战斗时长（秒）',
    rewards      VARCHAR(255) NOT NULL DEFAULT '' COMMENT '奖励描述，如 exp:100,gold:500',
    created_at   TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (uid) REFERENCES player (uid)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

			logger.Printf("战斗记录落库失败：code=%d msg=%s", resp.Code, resp.Msg)

		}
		logger.Printf("战斗记录落库成功：uid=%d exp+%d gold+%d", uid, resp.ExpGain, resp.GoldGain)
		return
	}

战斗的服务文件#

    DataSave 服务：内部服务，接收其他服务（如 battle）的写库请求。 

    内部协议沿用 common.Encode/Decode 的包格式，负载用 JSON（不对客户端暴露，省去 proto 代码生成）

    先定义几个MsgId

const (
	MsgSaveBattleRecord uint16 = 3000 // 写战斗记录并累加 exp/gold
	MsgDeductStamina    uint16 = 3001 // 扣体力
	MsgGetRole          uint16 = 3002 // 查角色

	msgSaveBattleRecordResp uint16 = 3100
	msgDeductStaminaResp    uint16 = 3101
	msgGetRoleResp          uint16 = 3102
)

   在链接上数据库后开始处理内部请求，然后就是handleConn解码、按MsgId分发、回包（和之前几篇的handleConn差不多就不花费篇幅了）。

// Serve 接受连接并处理内部请求；阻塞运行
func Serve(db *sql.DB, listener net.Listener) {
	s := &server{db: db}
	for {
		conn, err := listener.Accept()
		if err != nil {
			continue
		}
		go s.handleConn(conn)
	}
}

    parseRewards这个函数是 从 “exp:100”、“gold:500” 形式的奖励描述里解析累加值

func parseRewards(rewards []string) (exp, gold int32) {
	for _, r := range rewards {
		k, v, ok := strings.Cut(r, ":")
		if !ok {
			continue
		}
		n, err := strconv.Atoi(strings.TrimSpace(v))
		if err != nil {
			continue
		}
		switch strings.TrimSpace(k) {
		case "exp":
			exp += int32(n)
		case "gold":
			gold += int32(n)
		}
	}
	return exp, gold
}

    handleDeductStamina 函数是原子扣体力：stamina >= amount 才更新，不足则回包剩余体力。由于这里只用一条SQL语句所以就不用启动事务了。

	res, err := s.db.Exec(
		"UPDATE player SET stamina = stamina - ? WHERE uid = ? AND stamina >= ?",
		req.Amount, req.UID, req.Amount,
	)

		logger.Println("扣除体力失败:", err)

	n, err := res.RowsAffected()

    n, err := res.RowsAffected()是用于检测是否玩家存在且体力 ≥ 扣减量，如果是，则n=1，否则，n=0.

    如果n=0，就查询剩余体力并回包。

	if n == 0 {
		var stamina int32
		err := s.db.QueryRow("SELECT stamina FROM player WHERE uid = ?", req.UID).Scan(&stamina)
		if errors.Is(err, sql.ErrNoRows) {
			return &DeductStaminaResp{Code: codeNotFound, Msg: "玩家不存在"}
