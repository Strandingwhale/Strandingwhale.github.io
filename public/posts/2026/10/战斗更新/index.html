func pathLength() float64 {
	total := 0.0
	for i := 1; i < len(enemyPath); i++ {
		a, b := enemyPath[i-1], enemyPath[i]
		total += math.Hypot(b.X-a.X, b.Y-a.Y)
	}
	return total
}

    干员是否对敌人造成伤害检测

	for _, op := range r.operators {
		for _, e := range r.enemies {
			if e.HP <= 0 {
				continue
			}
			p := pointAtDistance(e.Dist)//pointAtDistance 按已行进距离求路径上的坐标；超出总长时返回终点
			if math.Hypot(p.X-float64(op.X), p.Y-float64(op.Y)) <= attackRange {
				e.HP -= attackDamage
			}
		}
	}

    其中pointAtDistance的函数是这样的，取相邻两段路径拐点，并计算距离。如果敌人已经前进的路程小于等于当前相邻拐点的距离，则计算当前坐标。如果大于，则减去相邻拐点的距离，继续检测下一个相邻拐点。

func pointAtDistance(d float64) Vec2 {
	for i := 1; i < len(enemyPath); i++ {
		a, b := enemyPath[i-1], enemyPath[i]
		seg := math.Hypot(b.X-a.X, b.Y-a.Y)
		if seg == 0 {
			return a
		}
		if d <= seg {
			t := d / seg
			return Vec2{X: a.X + (b.X-a.X)*t, Y: a.Y + (b.Y-a.Y)*t}
		}
		d -= seg
	}
	return enemyPath[len(enemyPath)-1]
}

    snapshotLocked 生成当前帧状态快照，调用方需持有 r.mu

func (r *Room) snapshotLocked() *api.BattleStatePush {
	s := &api.BattleStatePush{//房间状态
		RoomId: r.ID,
		Frame:  r.frame,
		Cost:   r.cost,
		BaseHp: r.baseHP,
	}
	for _, op := range r.operators {//干员状态
		s.Operators = append(s.Operators, &api.OperatorState{
			OperatorId: op.ID,
			GridX:      op.X,
			GridY:      op.Y,
			Hp:         operatorMaxHP,
			Direction:  op.Direction,
		})
	}
	for _, e := range r.enemies {//敌人状态
		p := pointAtDistance(e.Dist)
		s.Enemies = append(s.Enemies, &api.EnemyState{
			EnemyId:    enemyConfigID,
			InstanceId: e.InstanceID,
			Hp:         e.HP,
			PosX:       float32(p.X),
			PosY:       float32(p.Y),
		})
	}
	return s
}

    pushState 下发帧状态（msgId 2004）给客户端 

func (r *Room) pushState(state *api.BattleStatePush) {
	body, err := proto.Marshal(state)
	if err != nil {
		logger.Println("序列化 BattleStatePush 失败:", err)
		return
	}
	r.send(2004, body)
}

    结束后pushSettle 下发结算（msgId 2005），随后销毁房间

func (r *Room) pushSettle(result int32) {
	duration := int32(time.Since(r.startedAt).Seconds())
	score := int32(0)
	var rewards []string
	if result == resultWin {
		score = 1000
		rewards = []string{"exp:100", "gold:500"}
	}
	body, err := proto.Marshal(&api.BattleSettlePush{
		RoomId:      r.ID,
		Result:      result,
		Score:       score,
		DurationSec: duration,
		Rewards:     rewards,
	})
	if err != nil {
		logger.Println("序列化 BattleSettlePush 失败:", err)
		return
	}
	r.send(2005, body)
	logger.Printf("房间%d 战斗结束：result=%d 用时%ds 分数%d", r.ID, result, duration, score)
	removeRoom(r.ID)
}
